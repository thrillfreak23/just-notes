* CORRELATIVES

  $ lot
  *insapient interrogative pronoun :: what object ? which?*
  *lot :: l- + -ot (from tot :: ens, entity, specimen, object)*
  *which? what (one)? ::* 

  $ lûn
  *sapient interrogative pronoun :: who?*
  *lûn :: l- + -ûn :: special suffix meaning "a sapient"*
  *lûn :: who?*

  $ -ûn 
  *correlative suffix :: special suffix meaning "a sapient"*
  *anomalous suffix in that there is no nominal decapitated to form -ûn*
  *nor is -ûn usable as as independent word*
